cmake_minimum_required(VERSION 3.24)
project(dracula_x_pc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_subdirectory(src/saturnfs)
add_subdirectory(src/audio)
add_subdirectory(src/saturn)
add_subdirectory(src/sh2)
enable_testing()
add_subdirectory(tests)
add_executable(dxpc-extract src/tools/extract_main.cpp)
target_link_libraries(dxpc-extract PRIVATE saturnfs)
add_executable(dxpc-catalog src/tools/catalog_main.cpp)
target_link_libraries(dxpc-catalog PRIVATE saturnfs)
add_executable(dxpc-overlays src/tools/overlays_main.cpp)
target_link_libraries(dxpc-overlays PRIVATE saturnfs)
add_executable(dxpc-music-export src/tools/music_export_main.cpp)
target_link_libraries(dxpc-music-export PRIVATE saturnfs dxpc_audio)
add_executable(dxpc-run src/tools/run_main.cpp)
target_link_libraries(dxpc-run PRIVATE saturnfs saturn sh2)
